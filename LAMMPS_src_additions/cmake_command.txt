# example 2: A100 node on Delta cluster

/usr/local/bin/cmake \
    -B build_${LAMMPS_tag} \
    -S lammps-19Nov2024/cmake \
    -DBUILD_MPI=yes \
    -DBUILD_OMP=yes \
    -DLAMMPS_MACHINE=${LAMMPS_tag} \
    -DBUILD_SHARED_LIBS=yes \
    -DCMAKE_INSTALL_PREFIX=/usr/local \
    -DPKG_GPU=yes \
    -DPKG_KOKKOS=on \

    -DKokkos_ARCH_HOSTARCH=ZEN3 \
    -DKokkos_ENABLE_OMP=yes \
    -DKokkos_ENABLE_CUDA=yes \
    -DGPU_API=cuda \
    -DKokkos_ARCH_AMPERE80=yes \
    -DGPU_ARCH=sm_80 \
    -DCMAKE_CUDA_ARCHITECTURES=80 \
    -DPKG_OPENMP=yes \
    -DPKG_MOLECULE=yes \
    -DPKG_ASPHERE=yes \
    -DPKG_BROWNIAN=yes \
    -DPKG_RIGID=yes \
    -DPKG_CG-DNA=yes \
    -DPKG_EXTRA-MOLECULE=yes \
